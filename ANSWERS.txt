public class Solution {
    public static int tripleXorSum(int N,int[] AR){
        long sum = 0;
        long MOD = 1000000007;
        long[] setBits = new long[32];
        long[] twoPowers = new long[32];
 
        twoPowers[0] = 1;
        for (int i = 1; i < 32; i++) {
            twoPowers[i] = (2 * twoPowers[i-1]) % MOD;
        }
 
        for (int i : AR) {
            int temp = i;
            for (int j = 0; j < 32; j++) {
                if (temp%2 == 1) {
                    setBits[j]++;
                }
                temp = temp/2;
            }
        }
 
        for (int i = 0; i < 32; i++) {
            long ones = setBits[i];
            long zeroes = N - ones;
            long ones3 = ((ones * (ones-1) * (ones - 2))/6) % MOD;
            long ones1 = ((ones * zeroes * (zeroes-1))/2) % MOD;
 
            if (ones >= 3 && zeroes >= 2) {
                sum = (sum + (twoPowers[i] * (ones3 + ones1))) % MOD;
            } else if (zeroes < 2 && ones >= 3) {
                sum = (sum + (twoPowers[i] * ones3)) % MOD;
            } else if (ones < 3 && ones >= 1) {
                sum = (sum + (twoPowers[i] * ones1)) % MOD;
            }
        }
 
        return (int) (sum % MOD);
    }

public static int opQueue(int n, List<List<Integer>>v) {
        // Write your code here
         Queue<Integer> a = new LinkedList<>();
        Queue<Integer> b = new LinkedList<>();
        for(int i=0;i<n;i++){
            List<Integer> x=v.get(i);
            if(x.get(0)==1)a.add(x.get(1));
            else{
                Integer y=a.poll();
                if(x.get(1)==1)a.add(y);
                else b.add(y);
            }
        }
        long p=1,q=1;
        while(!a.isEmpty()){
            p=(p*a.poll()) % 1000000007;
        }
        while (!b.isEmpty()) {
            q = (q*b.poll()) % 1000000007;
        }
        return (int)(p+q)%1000000007;
        
    }

        }

        int totalElements=array.length;
        for(Map.Entry<Integer,Integer> entry:countMap.entrySet()){
            int freq=entry.getKey();
            int noOfEltsWithFreq=entry.getValue();
            int deletions=0;
            for(Map.Entry<Integer,Integer>innerEntry:countMap.entrySet()){
                int cfreq=innerEntry.getKey();
                int ccount=innerEntry.getValue();
                if(cfreq<freq){
                    deletions+=(cfreq*ccount);
                }
                else if(cfreq>freq){

            return count2;
        }

        }

public static int largestGroup(int N,int M,int[] AR){
        int[] remainderCount = new int[M];
 
        for(int ele : AR) {
            int remainder = ele % M;
            remainderCount[remainder]++; 
        }
 
        int cnt = 0;
 
        if(remainderCount[0] > 0) {
            cnt++;
        }
 
        for(int r=1;r <= M/2; r++){
            if(r == M-r){
                if(remainderCount[r] > 0){
                    cnt++;
                }
            } else {
                cnt += Math.max(remainderCount[r], remainderCount[M-r]);
            }
        }
        return cnt;
    }

5.REDISTRIBUTION
  public static int redistribution(int[] C){
        int seconds = 0;
        int mean = (Arrays.stream(C).sum()) / C.length;
        int remainder = (Arrays.stream(C).sum()) % C.length;
        int size  = C.length;

        Arrays.sort(C);
        
        if (remainder == 0) {
            for (int i : C) {
                seconds += (i > mean ? (i-mean) : (mean-i));
            }
        } else {
            for (int i = 0; i < (size - remainder); i++) {
                seconds += C[i] > mean ? (C[i] - mean) : (mean - C[i]);
            }
            for (int i = (size - remainder); i < size; i++) {
                seconds += C[i] > (mean+1) ? (C[i] - mean - 1) : (mean + 1 - C[i]);
            }
        }

        return seconds/2;
    }

   public static int computexor(List<Integer> arr,int[] lazy,int node){
        if(node>=arr.size()){
            return 0;
        }
        arr.set(node,arr.get(node)+lazy[node]);
        if(2*node+1<arr.size()) lazy[2*node+1]+=lazy[node];
        if(2*node+2<arr.size()) lazy[2*node+2]+=lazy[node];

        lazy[node]=0;
        int leftxor=computexor(arr,lazy,2*node+1);
        int rightxor=computexor(arr,lazy,2*node+2);
        return arr.get(node)^leftxor^rightxor;
    }
